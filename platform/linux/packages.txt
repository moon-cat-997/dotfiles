# Base packages installed by platform/linux/setup.sh via pacman.
# One name per line; '#' comments and blank lines are ignored.
#
# This is deliberately NOT shared with platform/macos/Brewfile: of the entries
# below, four are built into macOS, one has no macOS counterpart, and one is a
# different product entirely. A cross-OS name-mapping layer would be more code
# than the two lists it replaced.
#
# yay is NOT here on purpose — an unknown target makes pacman fail the whole
# transaction, and yay is AUR-only on vanilla Arch. setup.sh bootstraps it.

git
# git-hat (utilities/git-hat) needs both of these:
github-cli        # gitconfig delegates HTTPS credentials to `gh`
direnv            # loads each persona's .envrc on cd and UNloads it on the way out.
                  # The unload is the point: a CLI token leaking out of a persona
                  # directory would silently run `eas build` in a work project
                  # under the personal account.
zsh
stow
curl
wget
neovim
starship
xclip             # X11 clipboard bridge; macOS (pbcopy/pbpaste) has it built in
wl-clipboard      # the Wayland counterpart of xclip — nvim sets clipboard=unnamedplus,
                  # which silently misbehaves under Wayland with only xclip installed.
                  # Both are listed because the session type is not known at install time.

# Claude Code hooks and statusline-command.sh all shell out to node.
# Without this a fresh machine gets a Claude setup that errors on every hook.
nodejs
npm

# Neovim IDE toolchain (config in common/nvim, symlinked to ~/.config/nvim).
# Without these the config still loads, but core features fail — mostly silently,
# which is why each one is declared rather than left to whatever the machine has.
ripgrep           # project-wide text search; the fuzzy finder shells out to it
fd                # fast file listing for the file picker
fzf               # fuzzy matcher behind fzf-lua, LazyVim's default picker
lazygit           # git UI opened in a popup with <leader>gg
tree-sitter-cli   # REQUIRED: nvim-treesitter's main branch compiles parsers with it,
                  # and without it syntax highlighting fails to install with no error
clang             # provides clangd, the C++ language server
cmake             # C++ builds; clangd reads the compile_commands.json it generates

keyd              # system-wide key remapping; configs in platform/linux/keyd/
